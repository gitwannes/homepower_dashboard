# check largest gaps
python3 -c "
import sqlite3
c=sqlite3.connect('meter_history.db')
c.execute('PRAGMA temp_store=MEMORY')

def top_gaps(table, n=10):
  q=f'''
  WITH ordered AS (
    SELECT timestamp AS ts,
           LAG(timestamp) OVER (ORDER BY timestamp) AS prev_ts
    FROM {table}
  ),
  gaps AS (
    SELECT prev_ts AS gap_start, ts AS gap_end,
           (julianday(ts)-julianday(prev_ts))*24.0 AS hours
    FROM ordered
    WHERE prev_ts IS NOT NULL
  )
  SELECT gap_start, gap_end, ROUND(hours,2) AS hours
  FROM gaps
  ORDER BY hours DESC
  LIMIT {n}
  '''
  print('===', table, '===')
  for row in c.execute(q):
    print(row)

top_gaps('inverter_history_5min')
top_gaps('sensorbox_history_5min')
"
